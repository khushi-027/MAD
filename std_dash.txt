import 'package:flutter/material.dart';

class AssignmentsScreen extends StatelessWidget {
  const AssignmentsScreen({super.key});

  // Dataset from Practical 3
  static const List<Map<String, dynamic>> _assignments = [
    {
      'title': 'Flutter UI Design',
      'subject': 'MAD',
      'due': 'Due in 2 days',
      'status': 'Pending',
    },
    {
      'title': 'Database CRUD',
      'subject': 'DBMS',
      'due': 'Due in 5 days',
      'status': 'In Progress',
    },
    {
      'title': 'REST API Integration',
      'subject': 'MAD',
      'due': 'Submitted',
      'status': 'Completed',
    },
  ];

  Color _statusColor(String status) {
    switch (status) {
      case 'Completed':
        return const Color(0xFF2E7D32);
      case 'In Progress':
        return const Color(0xFFEF6C00);
      default:
        return const Color(0xFFC62828);
    }
  }

  IconData _statusIcon(String status) {
    switch (status) {
      case 'Completed':
        return Icons.check_circle;
      case 'In Progress':
        return Icons.hourglass_bottom;
      default:
        return Icons.pending_actions;
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Assignment List')),
      // ListView.builder: multi-child scrollable widget, handles any number
      // of assignments without overflowing the screen.
      body: ListView.builder(
        padding: const EdgeInsets.all(16.0),
        itemCount: _assignments.length,
        itemBuilder: (context, index) {
          final item = _assignments[index];
          final status = item['status'] as String;
          final color = _statusColor(status);

          return Card(
            margin: const EdgeInsets.only(bottom: 12),
            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
            child: Padding(
              padding: const EdgeInsets.all(14.0),
              child: Row(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  CircleAvatar(
                    radius: 22,
                    backgroundColor: color.withOpacity(0.12),
                    child: Icon(_statusIcon(status), color: color),
                  ),
                  const SizedBox(width: 14),
                  // Expanded: title/subject column stretches to fill remaining
                  // width so the layout adapts to any device size.
                  Expanded(
                    child: Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Text(
                          item['title'] as String,
                          style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 15),
                        ),
                        const SizedBox(height: 4),
                        Text(
                          '${item['subject']} • ${item['due']}',
                          style: const TextStyle(fontSize: 12, color: Colors.brown),
                        ),
                      ],
                    ),
                  ),
                  const SizedBox(width: 8),
                  // Flexible: status chip only takes the width its text needs
                  Flexible(
                    child: Container(
                      padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 6),
                      decoration: BoxDecoration(
                        color: color.withOpacity(0.12),
                        borderRadius: BorderRadius.circular(20),
                      ),
                      child: Text(
                        status,
                        style: TextStyle(color: color, fontSize: 11, fontWeight: FontWeight.bold),
                      ),
                    ),
                  ),
                ],
              ),
            ),
          );
        },
      ),
    );
  }
}

import 'package:flutter/material.dart';
import '../widgets/custom_widgets.dart';

class AttendanceScreen extends StatelessWidget {
  const AttendanceScreen({super.key});

  // Dataset from Practical 3
  static const List<Map<String, dynamic>> _attendance = [
    {'subject': 'DBMS', 'percent': 0.92},
    {'subject': 'MAD', 'percent': 0.88},
    {'subject': 'AI', 'percent': 0.95},
    {'subject': 'CN', 'percent': 0.90},
  ];

  @override
  Widget build(BuildContext context) {
    final double overall =
        _attendance.map((e) => e['percent'] as double).reduce((a, b) => a + b) / _attendance.length;

    return Scaffold(
      appBar: AppBar(title: const Text('Attendance Summary')),
      // ListView.builder: Multi-Child scrollable widget, ensures the screen
      // adapts to any device height without overflow.
      body: ListView(
        padding: const EdgeInsets.all(16.0),
        children: [
          // Single Child widget (Card) wrapping a Row/Column combination
          Card(
            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
            color: const Color(0xFF5D4037),
            child: Padding(
              padding: const EdgeInsets.all(20.0),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  const Text(
                    'Overall Attendance',
                    style: TextStyle(color: Colors.white70, fontSize: 14),
                  ),
                  const SizedBox(height: 6),
                  Text(
                    '${(overall * 100).toStringAsFixed(0)}%',
                    style: const TextStyle(
                      color: Colors.white,
                      fontSize: 34,
                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  const SizedBox(height: 10),
                  // Row + Expanded: fills all available horizontal space responsively
                  Row(
                    children: [
                      Expanded(
                        child: ClipRRect(
                          borderRadius: BorderRadius.circular(8),
                          child: LinearProgressIndicator(
                            value: overall,
                            minHeight: 8,
                            backgroundColor: Colors.white24,
                            valueColor: const AlwaysStoppedAnimation(Colors.white),
                          ),
                        ),
                      ),
                    ],
                  ),
                ],
              ),
            ),
          ),
          const SectionHeader(title: 'Subject-wise Attendance', icon: Icons.bar_chart),
          // Column of subject rows built from the dataset
          ..._attendance.map((item) => _AttendanceTile(
                subject: item['subject'] as String,
                percent: item['percent'] as double,
              )),
        ],
      ),
    );
  }
}

class _AttendanceTile extends StatelessWidget {
  final String subject;
  final double percent;

  const _AttendanceTile({required this.subject, required this.percent});

  Color get _statusColor {
    if (percent >= 0.90) return const Color(0xFF2E7D32); // Good
    if (percent >= 0.75) return const Color(0xFFEF6C00); // Warning
    return const Color(0xFFC62828); // Critical
  }

  @override
  Widget build(BuildContext context) {
    return Card(
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
      child: Padding(
        padding: const EdgeInsets.symmetric(horizontal: 16.0, vertical: 12.0),
        child: Row(
          children: [
            // Flexible: subject name shrinks/grows but never forces overflow
            Flexible(
              flex: 2,
              child: Text(
                subject,
                style: const TextStyle(fontWeight: FontWeight.w600, fontSize: 15),
              ),
            ),
            const SizedBox(width: 12),
            // Expanded: progress bar always fills remaining space regardless of screen width
            Expanded(
              flex: 4,
              child: ClipRRect(
                borderRadius: BorderRadius.circular(8),
                child: LinearProgressIndicator(
                  value: percent,
                  minHeight: 10,
                  backgroundColor: Colors.grey.shade200,
                  valueColor: AlwaysStoppedAnimation(_statusColor),
                ),
              ),
            ),
            const SizedBox(width: 12),
            Text(
              '${(percent * 100).toStringAsFixed(0)}%',
              style: TextStyle(fontWeight: FontWeight.bold, color: _statusColor),
            ),
          ],
        ),
      ),
    );
  }
}

class TimetableScreen extends StatefulWidget {

  @override

class _TimetableScreenState extends State<TimetableScreen> {
  int _selectedDay = 0;

  static const List<String> _days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri'];

  // Weekly schedule built from the Practical dataset (DBMS, MAD, AI, CN)
  static const Map<String, List<Map<String, String>>> _schedule = {
    'Mon': [
      {'time': '09:00 - 10:00', 'subject': 'DBMS', 'room': 'Room 204'},
      {'time': '10:00 - 11:00', 'subject': 'MAD', 'room': 'Room 105'},
      {'time': '11:15 - 12:15', 'subject': 'AI', 'room': 'Lab 3'},
    ],
    'Tue': [
      {'time': '09:00 - 10:00', 'subject': 'CN', 'room': 'Room 302'},
      {'time': '10:00 - 11:00', 'subject': 'DBMS', 'room': 'Lab 1'},
    ],
    'Wed': [
      {'time': '09:00 - 10:00', 'subject': 'AI', 'room': 'Room 204'},
      {'time': '10:00 - 11:00', 'subject': 'MAD', 'room': 'Lab 2'},
      {'time': '11:15 - 12:15', 'subject': 'CN', 'room': 'Room 105'},
    ],
    'Thu': [
      {'time': '09:00 - 10:00', 'subject': 'DBMS', 'room': 'Room 204'},
      {'time': '10:00 - 11:00', 'subject': 'AI', 'room': 'Lab 3'},
    ],
    'Fri': [
      {'time': '09:00 - 10:00', 'subject': 'MAD', 'room': 'Room 105'},
      {'time': '10:00 - 11:00', 'subject': 'CN', 'room': 'Room 302'},
      {'time': '11:15 - 12:15', 'subject': 'AI', 'room': 'Lab 1'},
    ],
  };

  static const Map<String, Color> _subjectColors = {
    'DBMS': Color(0xFF5D4037),
    'MAD': Color(0xFF8D6E63),
    'AI': Color(0xFF6D4C41),
    'CN': Color(0xFFA1887F),
  };

  @override
  Widget build(BuildContext context) {
    final entries = _schedule[_days[_selectedDay]] ?? [];

    return Scaffold(
      appBar: AppBar(title: const Text('Timetable')),
      body: Column(
        children: [
          // Wrap: multi-child widget that lays day chips out responsively,
          // moving to a new line automatically on narrow screens.
          Padding(
            padding: const EdgeInsets.all(12.0),
            child: Wrap(
              spacing: 8,
              children: List.generate(_days.length, (index) {
                final selected = index == _selectedDay;
                return ChoiceChip(
                  label: Text(_days[index]),
                  selected: selected,
                  onSelected: (_) => setState(() => _selectedDay = index),
                  selectedColor: const Color(0xFF5D4037),
                  labelStyle: TextStyle(
                    color: selected ? Colors.white : const Color(0xFF5D4037),
                    fontWeight: FontWeight.bold,
                  ),
                  backgroundColor: Colors.white,
                );
              }),
            ),
          ),
          // Expanded + ListView: remaining vertical space is filled and the
          // period list scrolls independently of the day-selector above it.
          Expanded(
            child: entries.isEmpty
                ? const Center(child: Text('No classes scheduled', style: TextStyle(color: Colors.brown)))
                : ListView.separated(
                    padding: const EdgeInsets.symmetric(horizontal: 16.0),
                    itemCount: entries.length,
                    separatorBuilder: (_, __) => const SizedBox(height: 10),
                    itemBuilder: (context, index) {
                      final entry = entries[index];
                      final color = _subjectColors[entry['subject']] ?? const Color(0xFF5D4037);
                      return _PeriodTile(
                        time: entry['time']!,
                        subject: entry['subject']!,
                        room: entry['room']!,
                        color: color,
                      );
                    },
                  ),
          ),
        ],
      ),
    );
  }
}
